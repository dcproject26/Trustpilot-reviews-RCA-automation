async function fetchInbox() {
  try {
    const rows = await (await fetch('/api/reviews')).json();
    const byId = new Map(REVIEWS.map(r => [r.id, r]));
    const merged = rows.map(row => {
      const demo = byId.get(row.id);
      if (demo) {
        return {
          ...demo,
          receivedAt:      row.received_at || demo.receivedAt,
          status:          row.status || demo.status,
          matchTier:       row.match_tier ?? demo.matchTier,
          candidateState:  row.candidate_state ?? demo.candidateState,
        };
      }
      // New API-only review (manual / Slack): minimal template
      return {
        id:              row.id,
        author:          row.author || 'Unknown',
        rating:          row.rating || 1,
        language:        row.language || 'en',
        receivedAt:      row.received_at,
        status:          row.status || 'new',
        matchTier:       row.match_tier,
        candidateState:  row.candidate_state || false,
        experience:      row.experience || '',
        originalText:    row.snippet || '',
        englishText:     row.snippet || '',
        booking:         {},
        rca:             {},   // loadDraftOverlays populates from /api/reviews/{id}
      };
    });
    merged.sort((a, b) => new Date(b.receivedAt) - new Date(a.receivedAt));
    REVIEWS.length = 0;
    REVIEWS.push(...merged);
  } catch (e) {
    console.warn('[inbox] fetch failed, keeping demo REVIEWS:', e);
  }
}
